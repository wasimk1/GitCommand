# 4. SAVE CHANGES IN GIT
# =====================================
git status

git push origin practice/commands

# =====================================
# 5. MERGE INTO MAIN
# =====================================
git switch main
git pull origin main
git merge practice/commands
git push origin main

# =====================================
# 6. AGAIN SWITCH BACK TO WORKING BRANCH
# =====================================
git switch practice/commands
git status
take latest from main branch : git pull origin main
